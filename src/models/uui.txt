Почему cleanupScheduler так шумит
Судя по логам, scheduler запускается по таймеру и на каждом тике делает запрос в Mongo.
Когда Mongo недоступна, он каждый раз логирует full stack trace.

Это норм в dev, но в проде лучше сделать мягче.

Что я бы улучшил в cleanupScheduler
1. Не падать, а логировать кратко
2. Добавить защиту от overlapping runs
3. Добавить backoff / cooldown при Mongo network errors
Пример улучшенного scheduler
Вот аккуратная версия:

js

let isCleanupRunning = false;
let mongoFailureCount = 0;

const isMongoNetworkError = (error) => {
  const message = error?.message || "";
  const code = error?.code || "";
  const causeCode = error?.cause?.code || "";

  return (
    code === "ECONNABORTED" ||
    message.includes("MongoServerSelectionError") ||
    message.includes("ENOTFOUND") ||
    causeCode === "ENOTFOUND"
  );
};

const runCleanupSafely = async () => {
  if (isCleanupRunning) {
    console.warn("[cleanupScheduler] skip: previous cleanup still running");
    return;
  }

  isCleanupRunning = true;

  try {
    await cleanupExpiredJobs();
    mongoFailureCount = 0;
  } catch (error) {
    if (isMongoNetworkError(error)) {
      mongoFailureCount += 1;
      console.warn(
        `[cleanupScheduler] Mongo temporarily unavailable (${mongoFailureCount}): ${error.message}`,
      );
    } else {
      console.error("[cleanupScheduler] cleanup error:", error);
    }
  } finally {
    isCleanupRunning = false;
  }
};

Этап 4. Решить судьбу persistent storage
Тут уже два режима.

Режим A — совместимый
Оставляешь storage/ для финальных файлов.
Это безопасный переход.

Режим B — strict tmpfs
Финальные файлы тоже только в workspace/tmpfs, и после TTL исчезают.
Но это уже следующий шаг, не первый.

Как сделать это безболезненно в коде
Шаг 1. Ввести единый path-helper
Сделай модуль, например:

js

// src/utils/paths.js
const path = require("path");
const os = require("os");

const PROJECT_ROOT = path.resolve(__dirname, "../..");
const STORAGE_ROOT = path.join(PROJECT_ROOT, "storage");
const TMP_ROOT = path.join(os.tmpdir(), "armia-doc");

const getJobWorkspaceDir = (jobId) => path.join(TMP_ROOT, `job-${jobId}`);

const getJobWorkspaceFile = (jobId, fileName) =>
  path.join(getJobWorkspaceDir(jobId), fileName);

module.exports = {
  PROJECT_ROOT,
  STORAGE_ROOT,
  TMP_ROOT,
  getJobWorkspaceDir,
  getJobWorkspaceFile,
};
Шаг 2. Перед генерацией создавать workspace
Например:

js

const fs = require("fs/promises");
const { getJobWorkspaceDir } = require("../utils/paths");

const ensureJobWorkspace = async (jobId) => {
  const dir = getJobWorkspaceDir(jobId);
  await fs.mkdir(dir, { recursive: true });
  return dir;
};
Шаг 3. runProfiles перевести на workspace path
Вместо:

js

const buildDocxPath = (job) =>

const path = require("path");

Шаг 4. После успешной генерации копировать в storage
Например:

js

const copyFinalArtifactsToStorage = async (job, workspaceDocxPath, workspacePdfPath) => {
  const docxStoragePath = path.join(STORAGE_ROOT, "docx", `${job._id}.docx`);
  const pdfStoragePath = path.join(STORAGE_ROOT, "pdf", `${job._id}.pdf`);

  await fs.copyFile(workspaceDocxPath, docxStoragePath);
  await fs.copyFile(workspacePdfPath, pdfStoragePath);

  return {
    docx: `${job._id}.docx`,
    pdf: `${job._id}.pdf`,

};

Кроме удаления storage/... добавить:

force: true — хорошо для tmp cleanup.

Почему это безболезненно
Потому что:

фронт ничего не замечает
job.files остаётся тем же
старый download flow не ломается
cleanup просто становится умнее
можно откатить в любой момент
Что важно учесть для Windows
Ты сейчас, судя по логам, работаешь на Windows:

txt

C:\Users\Александр\Documents\GitHub\armia-doc
tmpfs как Linux tmpfs на Windows нет в том же виде.

Что это значит:
На локальной Windows-разработке ты сможешь использовать:

js

os.tmpdir()
но это будет просто системная temp-директория, не RAM tmpfs.

Реальный tmpfs only будет актуален:
на Linux сервере
в Docker/Kubernetes с tmpfs mount
в WSL/Linux VM
То есть сейчас можно сделать архитектуру под tmp workspace, а “настоящий tmpfs” включить уже на сервере.
